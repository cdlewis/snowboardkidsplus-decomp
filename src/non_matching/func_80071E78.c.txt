s32 func_80071E78(TextureColor *color, s8 *stepPtr, u8 minimum, u8 maximum) {
    s32 reversed = 0;
    s32 step = *stepPtr;
    u8 value = color->r;
    s8 nextStep = step;

    if (step > 0) {
        if (value + step > maximum) {
            value = maximum;
            goto reverse; 
        } else {
            value += nextStep;
        }
    } else if (step < 0) {
        if (value - minimum < -step) {
            value = minimum;
reverse:
            nextStep = -nextStep;
            reversed = 1;
        } else {
            value += nextStep;
        }
    }
    color->r = color->g = color->b = value;
    *stepPtr = nextStep;
    return reversed;
}
