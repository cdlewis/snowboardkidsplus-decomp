typedef struct TextureRegion {
    u8 u, v, w, h;
} TextureRegion;

typedef struct TextureRegionSet {
    s16 x, y;
    TextureRegion *regions;
    s16 depth;
    s16 clutX, clutY;
    s16 unk0E;
} TextureRegionSet;

typedef struct DrawModeSpritePacket {
    u32 tag;
    u16 tpage, drawModeCode;
    u8 r0, g0, b0, code;
    s16 x0, y0;
    u8 u0, v0;
    u16 clut;
    s16 w, h;
} DrawModeSpritePacket;

extern TextureRegionSet D_8012DDA4[];
void func_8006C998(s16 x, s16 y, s32 index, s16 palette, s32 blendMode, GsOT *ot, u16 depth) {
    TexturedQuadFrame *frame = &D_8012DFF4[index];
    TextureRegionSet *texture = &D_8012DDA4[frame->textureIndex];
    TextureRegion *region = &texture->regions[frame->frameIndex];
    DrawModeSpritePacket *sprite = (DrawModeSpritePacket *)GsGetWorkBase();
    sprite->tag = (sizeof(*sprite) / sizeof(u32) - 1) << 24;
    sprite->drawModeCode = 0xE100;
    setRGB0(sprite, gTexturedSpriteTemplate.r0, gTexturedSpriteTemplate.g0, gTexturedSpriteTemplate.b0);
    if ((s16)blendMode >= 0) {
        u16 tp = texture->depth;
        s16 ty = texture->y;
        u16 tx = texture->x;
        sprite->code = 0x64 | 2;
        sprite->tpage = getTPage(tp, blendMode, tx, ty);
    } else {
        u16 tp = texture->depth;
        s16 ty = texture->y;
        u16 tx = texture->x;
        sprite->code = 0x64;
        sprite->tpage = getTPage(tp, 0, tx, ty);
    }
    {
    u16 clut = getClut(texture->clutX, texture->clutY + palette);
    sprite->x0 = x;
    sprite->y0 = y;
    sprite->clut = clut;
    }
    sprite->u0 = region->u;
    sprite->v0 = region->v;
    sprite->w = region->w;
    sprite->h = region->h;
    AddPrim(&ot->org[depth], sprite);
    func_800BF2C8((u32)(sprite + 1));
}
