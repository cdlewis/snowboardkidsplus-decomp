typedef struct AnimationResource {
    u8 unk00[0xC6];
    s16 animationLimit;
    s16 animationStep;
    s16 animationPosition;
    u8 unkCC[0x34];
} AnimationResource;

s32 func_80059290(AnimationResource *resource) {
    resource->animationPosition += resource->animationStep;
    if (resource->animationPosition < resource->animationLimit) {
        return 1;
    }
    resource->animationPosition = (resource->animationPosition - resource->animationStep) & 0xFF00;
    return 0;
}
