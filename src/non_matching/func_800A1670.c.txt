typedef struct LoadedTextDescriptor {
    u16 *text;
    u16 type;
    u16 width;
} LoadedTextDescriptor;

void func_800A1670(void) {
    LoadedTextDescriptor *entry = (LoadedTextDescriptor *)0x801F2000;
    u16 *text = (u16 *)0x801F2080;
    LoadedTextDescriptor *current;
    s16 i;
    for (i = 0; i < 16; entry++) {
        current = entry;
        if (*text == 0x807A) {
            current->text = 0;
            current->type = 2;
            current->width = 0;
            return;
        }
        if (*text == 0x8074) {
            text++;
            current->text = text;
            current->type = 0;
        } else {
            current->text = text;
            current->type = 1;
        }
        current->width = 0;
        for (;;) {
            u16 ch = *text;
            if (ch == 0x800A) {
                text++;
                continue;
            }
            if (ch == 0x8065) {
                *text = 0;
                break;
            }
            if (ch == 0x8020 || !(ch & 0xC000)) {
                current->width++;
                text++;
            } else {
                text++;
            }
        }
        text++;
        while (*text++ == 0) {}
        text--;
        i++;
        if (i >= 16) break;
    }
}
